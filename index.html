async function loadPublicCatalog(codeOrId) { try { show("publicCatalogScreen"); setText("publicCatalogTitle", "Cargando catálogo..."); let bData = null; let bId = null; const localList = JSON.parse(localStorage.getItem("stockea_local_businesses") || "[]"); const foundLocal = localList.find(b => b.id === codeOrId || b.accessCode === String(codeOrId).toUpperCase()); if (foundLocal) { bData = foundLocal; bId = foundLocal.id; products = JSON.parse(localStorage.getItem(`stockea_local_products_${bId}`) || "[]"); } else { try { // Intentar buscar directamente por ID de documento en Firestore let snap = await db.collection("businesses").doc(codeOrId).get(); if (snap.exists) { bData = snap.data(); bId = snap.id; } else { // Si no es por ID, buscar por el código de acceso const query = await db.collection("businesses").where("accessCode", "==", String(codeOrId).toUpperCase()).get(); if (!query.empty) { snap = query.docs[0]; bData = snap.data(); bId = snap.id; } } if (bId) { const prodSnap = await db.collection("businesses").doc(bId).collection("products").get(); products = prodSnap.docs.map(doc => ({ id: doc.id, ...doc.data() })); } } catch (fbError) { console.warn("Error leyendo de Firestore:", fbError); } } if (!bData) { setText("publicCatalogTitle", "Catálogo no encontrado"); $("publicProductList").innerHTML = '
El catálogo que buscás no existe o el enlace es incorrecto. Verificá que el negocio esté creado en la nube.
'; return; } currentBusiness = { id: bId, ...bData }; setText("publicCatalogTitle", currentBusiness.name || "Catálogo"); setText("publicCatalogSub", `Contacto / WhatsApp: ${currentBusiness.phone || "No especificado"}`); fillPublicCategories(); renderPublicProducts(); } catch (error) { console.error(error); alert("Error al cargar el catálogo público."); } }
